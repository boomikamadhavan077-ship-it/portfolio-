WITH MonthlySales AS (
  SELECT 
    DATE_TRUNC('month', transaction_date) AS month,
    region,
    SUM(revenue) AS total_revenue,
    COUNT(DISTINCT customer_id) AS active_clients
  FROM enterprise_sales
  GROUP BY 1, 2
)
SELECT 
  month, region, total_revenue,
  AVG(total_revenue) OVER (PARTITION BY region) AS avg_regional_rev
FROM MonthlySales
ORDER BY total_revenue DESC;

    df = pd.read_csv(filepath)
    df.dropna(subset=['transaction_id', 'amount'], inplace=True)
    df['amount'] = df['amount'].astype(float)
    df['date'] = pd.to_datetime(df['date'])
